cmake_minimum_required(VERSION 3.16)
project(EmbeddedEmulator C)
set(CMAKE_C_STANDARD 11)
add_library(cpu STATIC cpu.c)
target_include_directories(cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(emulator main.c)
target_link_libraries(emulator PRIVATE cpu)
enable_testing()
add_executable(cpu_tests tests.c)
target_link_libraries(cpu_tests PRIVATE cpu)
add_test(NAME cpu_tests COMMAND cpu_tests)
